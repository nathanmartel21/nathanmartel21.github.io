# OPTIONNEL — améliore juste les types de champs (date, keyword, ip).
# ⚠️ Sur Bonsai gratuit/Hobby, `_index_template` renvoie 403 "Forbidden action" :
#    c'est bloqué par leur plan. Dans ce cas, IGNORE ce fichier — les index se
#    créent tout seuls au premier document (mapping dynamique), l'app marche pareil.
#    (Alternative si tu veux typer le champ ip : crée l'index à la main, ex.
#     PUT /coffre-events  { "mappings": { "properties": { "ip": {"type":"ip"} } } }
#     — si ça renvoie aussi 403, laisse tomber, ce n'est pas nécessaire.)
#
# Sinon (Elasticsearch self-host / Elastic Cloud) : à coller dans Dev Tools.

PUT _index_template/access-logs
{
  "index_patterns": ["access-logs*"],
  "template": { "mappings": { "properties": {
    "@timestamp": { "type": "date" },
    "event":      { "type": "keyword" },
    "email":      { "type": "keyword" },
    "role":       { "type": "keyword" },
    "status":     { "type": "keyword" },
    "by":         { "type": "keyword" },
    "ip":         { "type": "ip" }
  } } }
}
